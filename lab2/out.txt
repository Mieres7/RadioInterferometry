def __DW_IO__(session_id, async_type=None):
    from builtins import dict, Exception, isinstance, object, str

    DW_OUTPUT_MIME_TYPE = "application/vnd.microsoft.datawrangler.output+json"

    def comm():
        from comm import create_comm
        comm = create_comm("datawrangler")
        return lambda msg: comm.send(msg, {"async": True})
    
    def display():
        from IPython.display import DisplayHandle
        handle = DisplayHandle()
        handle.display({ DW_OUTPUT_MIME_TYPE: {} }, raw=True)
        return lambda msg: handle.update({ DW_OUTPUT_MIME_TYPE: msg }, raw=True, metadata={"async": True})

    types = {
        "comm": comm,
        "display": display,
    }

    send_async_output = None
    if async_type in types:
        try:
            send_async_output = types[async_type]()
        except Exception:
            pass
    
    def output_mime(data):
        import sys
        class MimeBundle(object):
            def __init__(self, data):
                self.data = data
            def _repr_mimebundle_(self, **kwargs):
                return { "text/plain": "", **self.data }
            # 'text/plain' may be added to the bundle automatically. Make sure it does not take up space.
            def __str__(self):
                return ""
            def __repr__(self):
                return ""
        sys.displayhook(MimeBundle(data))
    
    def output(data):
        # vscode-jupyter has a problem handling non-object JSON at the top level. So make sure we never send it.
        if not isinstance(data, dict):
            raise Exception("Output data must be a dictionary")
        output_mime({ DW_OUTPUT_MIME_TYPE: data })

    def output_async(**data):
        msg = {"_session": session_id, **data}
        if send_async_output:
            send_async_output(msg)
        
    def log_exception(error_details=None, props={}, measures={}):
        if error_details is not None:
            errorName = error_details["name"]
            error = error_details["message"]
        else:
            import sys
            (errorName, error) = sys.exc_info()[:2]
        output_async(type="error", name=str(errorName), message=str(error), properties=props, measures=measures)
    
    def log_telemetry(name, props={}, measures={}):
        output_async(type="telemetry", event=name, properties=props, measurements=measures)

    return dict(
        output=output,
        output_async=output_async,
        log_exception=log_exception,
        log_telemetry=log_telemetry,
    )

def __DW_PANDAS_TRANSPORT__():
    from builtins import any, AttributeError, bool, dict, Exception, hasattr, ImportError, isinstance, len, list, map, min, object, range, str, tuple, type

    def get_df_shape(df):
        shape = {}
        shape["rows"], shape["columns"] = df.shape
        return shape

    def get_index_column_name(df):
        try:
            index_column_name = df.index.name if df.index.name else str(tuple(df.index.names)) if len(df.index.names) > 1 else "index"
        except AttributeError:
            index_column_name = "index"
        return index_column_name

    def get_inferred_dtype(column):
        import pandas as pd
        # In some older versions of Pandas, infer_dtype does not support sparse columns
        if hasattr(column, "sparse"):
            column = column.sparse.to_dense()
        inferred_dtype = pd.api.types.infer_dtype(column, skipna=True)
        return inferred_dtype

    def map_dtype_to_dw_type(dtype, inferred_dtype=None):
        """
        References to known types:
        - Pandas: https://stackoverflow.com/a/29246498
        - NumPy: https://numpy.org/doc/stable/reference/arrays.scalars.html

        Note: the types here do not necessarily include all NumPy types, but rather
        what type Pandas uses when the type is used as input. For example, NumPy
        supports several precision for datetimes, while Pandas internally only uses
        nanoseconds, see https://github.com/pandas-dev/pandas/issues/6741#issuecomment-39026803

        Note that the value mapped here is used for the following purposes:
        - To figure out what "picker" should be used for this type in the UI, e.g. when replacing
        - To figure out what operations should be associated with this type in the UI
        """
        import re

        # string types
        if dtype == "string" or (dtype == "object" and inferred_dtype == "string"):
            return "string"

        # category types
        # TODO@DW: enable this
        # if dtype == "category":
        #     return "category"

        # float types
        # TODO@DW: consider if we need to distinguish nullable types
        # or pass it along as metadata for certain operations
        if re.fullmatch(r"(?:float|Float)\d+", dtype):
            return "float"

        # int types
        # TODO@DW: consider if we need to distinguish nullable types
        # or pass it along as metadata for certain operations
        if re.fullmatch(r"(?:int|uint|Int|UInt)\d+", dtype):
            return "integer"

        # complex types
        # TODO@DW: enable this
        # if re.fullmatch("complex\d+", dtype):
        #     return "complex"

        # boolean types
        # TODO@DW: consider if we need to distinguish nullable types
        # or pass it along as metadata for certain operations
        if dtype in ["bool", "boolean"]:
            return "boolean"

        # datetime types
        # TODO@DW: handle timezone-specific datetimes
        if dtype == "datetime64[ns]":
            return "datetime"

        # timedelta types
        # TODO@DW: enable this
        # if dtype == "timedelta64[ns]":
        #     return "timedelta"

        # period types
        # TODO@DW: enable this
        # if re.fullmatch(r"period[.*]", dtype):
        #    return "period"

        # interval types
        # TODO@DW: we will need to pass the metadata of the interval type as well
        # interval_match = re.fullmatch(r"interval[(.*), .*]", dtype)
        # if interval_match:
        #    return map_dtype_to_dw_type(interval_match.group(1))

        # sparse types

        # TODO@DW: sparse types don't well for us because they don't support assignments,
        # e.g. https://github.com/pandas-dev/pandas/issues/21818 or describe calls.
        # For now we can just render them, but not support operations on them yet.

        # for sparse types, they're essentially just more compact versions of the
        # type that they wrap around, so we can just extract that part instead
        # sparse_match = re.fullmatch(r"(?:Sparse\[(.*), .*\]|Sparse\[(.*)\])", dtype)
        # if sparse_match:
        #     return map_dtype_to_dw_type(sparse_match.group(1))

        # default to unknown
        return "unknown"

    def is_missing(value):
        import pandas as pd

        return pd.api.types.is_scalar(value) and bool(pd.isna(value))

    def serialize(x):
        if is_missing(x):
            return None
        return str(x)

    def serialize_row(df, idx):
        key = df.index[idx]
        data = [serialize(key), *(serialize(df.iat[idx, col]) for col in range(len(df.columns)))]
        return data

    def pd_dumps(data):
        try:
            # This was renamed in Pandas 2.2.0. We need to try both.
            from pandas.io.json import ujson_dumps as pd_dumps
        except ImportError:
            from pandas.io.json import dumps as pd_dumps

        return pd_dumps(data, iso_dates=True, default_handler=str)
    
    def get_df_payload_v0(df, row_limit, **extra_props):
        import pandas as pd
        data = []
        for i in range(0, min(len(df), row_limit)):
            data.append(serialize_row(df, i))
        column_types = [df.index.dtype] + list(df.dtypes)
        column_names = [get_index_column_name(df)] + list(df)
        columns = [df.index] + [df.iloc[:, column_idx] for column_idx in range(len(list(df)))]
        column_info_list = []
        for column_idx in range(0, len(column_names)):
            column_info = {}

            # for each column, figure out the type and name
            raw_column_type = str(column_types[column_idx])
            column_name = column_names[column_idx]
            column = columns[column_idx]
            column_info["name"] = str(column_name)
            column_info["rawType"] = raw_column_type
            column_info["type"] = map_dtype_to_dw_type(raw_column_type, "string" if pd.api.types.is_string_dtype(column) else None)
            column_info_list.append(column_info)

        return pd_dumps({
            **extra_props,
            "shape": get_df_shape(df),
            "rows": data,
            "columns": column_info_list
        })

    def get_df_payload(df, row_limit, **extra_props):
        return get_df_payload_v0(df, row_limit, **extra_props)

    # namespace exports
    return dict(
        is_missing=is_missing,
        get_df_shape=get_df_shape,
        get_index_column_name=get_index_column_name,
        get_inferred_dtype=get_inferred_dtype,
        map_dtype_to_dw_type=map_dtype_to_dw_type,
        serialize=serialize,
        serialize_row=serialize_row,
        pd_dumps=pd_dumps,
        get_df_payload=get_df_payload
    )

def __DW_OUTPUT_FORMATTER__(api):
    import IPython
    import traitlets
    import traceback
    from builtins import classmethod, dict, Exception, ImportError, list, set, str, type

    DW_JSON_MIME_TYPE = 'application/vnd.microsoft.datawrangler.viewer.v0+json'
    max_num_rows_to_preload = 50
    tmp_vars = api["output_cache"]
    io = api["io"]
    supported_types = set()

    class DataWranglerFormatter(IPython.core.formatters.BaseFormatter):
        format_type = traitlets.Unicode(DW_JSON_MIME_TYPE)
        print_method = traitlets.ObjectName('_repr_dw_')

    class DataWrangler(IPython.display.DisplayObject):
        """
        Data Wrangler formatter.
        """

        is_supported = False

        def __init__(self, expr_val):
            import time
            start_time = time.time()
            props = {}

            try:
                nonlocal supported_types
                type_str = type_to_str(type(expr_val))

                pandas_supported = is_pandas_supported()
                if pandas_supported and type_str in supported_types:
                    pandas_df, conversion_method = api["pandas_converter"](type_str, expr_val)
                    self.is_supported = pandas_df is not None

                # Base telemetry props
                props = { "type": type_str, "conversionMethod": conversion_method, "supportedTypes": list(supported_types) }

                if self.is_supported:
                    import uuid
                    self.id = str(uuid.uuid4())

                    # create a shallow copy in case a displayed object is mutated in the same cell
                    if pandas_df is expr_val:
                        pandas_df = pandas_df.copy(deep=False)

                    tmp_vars[self.id] = {
                        "original": expr_val,
                        "converted": pandas_df,
                        "conversion_method": conversion_method,
                        "original_type": type_str,
                    }

                    io["log_telemetry"](name="OutputConverted", props=props, measures={"conversionTime": time.time() - start_time})
                elif type_str in supported_types:
                    props["isPandasSupported"] = pandas_supported
                    io["log_exception"](
                        error_details={
                            "name": "OutputConversionError",
                            "message": "Failed to convert"
                        },
                        props=props,
                        measures={"conversionTime": time.time() - start_time}
                    )
                else:
                    # This can be called for subclasses, so we don't actually want to log them to avoid PII.
                    # We just want to check here if we're failing because we don't have pandas or it's a subtype.
                    del props["type"]
                    props["isPandasSupported"] = pandas_supported
                    io["log_telemetry"](name="OutputConversionSkipped", props=props, measures={"conversionTime": time.time() - start_time})
            except Exception:
                # We shouldn't receive errors here, but if we do we can send telemetry for it.
                io["log_exception"](
                    props=props,
                    measures={"conversionTime": time.time() - start_time}
                )

        @classmethod
        def formatter(cls, var, **kwargs):
            s = cls(var, **kwargs)
            return s._repr_dw_()

        def _repr_mimebundle_(self):
            mime_bundle = {}
            try:
                if self.is_supported:
                    mime_bundle[DW_JSON_MIME_TYPE] = self._gen_json()
            except Exception:
                # need to catch and print exception since it is user visible
                traceback.print_exc()
            return mime_bundle

        def _repr_dw_(self):
            try:
                return self._gen_json()
            except Exception:
                # need to catch and print exception since it is user visible
                traceback.print_exc()

        def _gen_json(self):
            if self.is_supported:
                kwargs = {"ref": self.id}
                return api["pandas_transport"]["get_df_payload"](tmp_vars[self.id]["converted"], max_num_rows_to_preload, **kwargs)

    def register_dw_mimetype():
        """Register _repr_dw_ with the IPython display mechanism."""
        shell = IPython.get_ipython()
        if not shell:
            return
        display_formatter = shell.display_formatter
        display_formatter.formatters.setdefault(
            DW_JSON_MIME_TYPE,
            DataWranglerFormatter(parent=display_formatter),
        )

    def enable_formatter_for_types(*args):
        """
        Enables Data Wrangler as the default IPython formatter for the given type.
        """
        nonlocal supported_types
        supported_types = set()
        register_dw_mimetype()
        formatters = IPython.get_ipython().display_formatter.formatters
        for arg in args:
            supported_types.add(type_to_str(arg) if type(arg) == type else arg)
            if type(arg) == str:
                try:
                    path, base = arg.rsplit(".", 1)
                except Exception:
                    path = "builtins"
                    base = arg
                formatters[DW_JSON_MIME_TYPE].for_type_by_name(
                    path, base, DataWrangler.formatter
                )
            else:
                formatters[DW_JSON_MIME_TYPE].for_type(
                    arg, DataWrangler.formatter
                )

    def disable_formatter():
        """
        Disables the formatter.
        """
        nonlocal supported_types
        formatters = IPython.get_ipython().display_formatter.formatters
        supported_types = []
        for key in list(formatters.keys()):
            # Only explicitly disable the MIME type we are registered with.
            # This will allow us to run with several MIME type versions concurrently.
            if key == DW_JSON_MIME_TYPE:
                formatters.pop(key)

    def type_to_str(type):
        return f'{type.__module__}.{type.__qualname__}'

    def is_pandas_supported():
        try:
            import pandas as _
            return True
        except ImportError:
            return False

    # namespace exports
    return dict(
        enable_for_types=enable_formatter_for_types,
        disable=disable_formatter
    )
